export async function productConfig(name = "local") {
  if (!/^[a-z][a-z0-9-]{0,31}$/.test(name))
    throw Error(
      "Installation name must be 1-32 lower-case letters, digits or hyphens",
    );
  const dir = join(root, ".agentbrake-local", "product-" + name);
  const config = JSON.parse(
    await readFile(join(dir, "installation.json"), "utf8"),
  );
  if (
    config.name !== name ||
    !/^\d+$/.test(String(config.port)) ||
    config.port < 1024 ||
    config.port > 65535 ||
    !/^[a-f0-9-]{36}$/.test(config.id)
  )
    throw Error("Invalid local installation manifest");
  const images = JSON.parse(
    await readFile(join(root, "infra/local/images.lock.json"), "utf8"),
  ).images;
  return {
    name,
    dir,
    id: config.id,
    port: config.port,
    project: "agentbrake-product-" + name,
    env: {
      ...process.env,
      COMPOSE_PROJECT_NAME: "agentbrake-product-" + name,
      INSTALLATION_ID: config.id,
      INSTALL_DIR: dir.replaceAll("\\", "/"),
      DASHBOARD_PORT: String(config.port),
      NODE_IMAGE: images.node.image,
      POSTGRES_IMAGE: images.postgres.image,
      OPA_IMAGE: images.opa.image,
      NEXT_TELEMETRY_DISABLED: "1",
    },
  };
}

export async function prepareProduct(config) {
  const hash = createHash("sha256");
  for (const file of ["agentbrake.rego", "authz.rego"])
    hash.update(await readFile(join(root, "services/policy", file)));
  const token = (await readFile(join(config.dir, "opa-token"), "utf8")).trim();
  await writeFile(
    join(config.dir, "opa-auth.json"),
    JSON.stringify({
      agentbrake_auth: { token },
      agentbrake_meta: { policy_hash: hash.digest("hex") },
    }),
    { mode: 0o600 },
  );
}
export async function dockerProduct(
  config,
  args,
  { capture = false, input, timeoutMs = 300000 } = {},
) {
  const dockerConfig = join(root, ".agentbrake-local/docker-public");
  await mkdir(dockerConfig, { recursive: true });
  const prefix = [
    "--config",
    dockerConfig,
    ...(process.platform === "win32"
      ? ["--host", "npipe:////./pipe/docker_engine"]
      : []),
  ];
  return new Promise((resolveResult, reject) => {
    let output = "",
      errorOutput = "";
    const child = spawn("docker", [...prefix, ...args], {
      cwd: root,
      env: config.env,
      windowsHide: true,
      stdio: [
        input === undefined ? "ignore" : "pipe",
        capture ? "pipe" : "inherit",
        capture ? "pipe" : "inherit",
      ],
    });
    const timer = setTimeout(() => {
      child.kill();
      reject(Error("Local Docker command deadline exceeded"));
    }, timeoutMs);
    child.stdout?.on("data", (chunk) => {
      output += chunk;
      if (output.length > 4000000) child.kill();
    });
    child.stderr?.on("data", (chunk) => {
      errorOutput += chunk;
      if (errorOutput.length > 4000000) child.kill();
    });
    child.on("error", (error) => {
      clearTimeout(timer);
      reject(error);
    });
    child.on("close", (code) => {
      clearTimeout(timer);
      if (code !== 0)
        reject(
          Object.assign(Error("Local Docker command failed (" + code + ")"), {
            output,
            errorOutput,
          }),
        );
      else resolveResult(output);
    });
    if (input !== undefined) child.stdin.end(input);
  });
}

async function main() {
  const [command = "status", name = "local", ...extra] = process.argv.slice(2);
  if (command === "init") {
    const config = await initializeProduct(name, Number(extra[0] ?? 4310));
    console.log(
      "Created local fixture installation " +
        config.name +
        ". Operator login keys: " +
        join(config.dir, "operators.json"),
    );
    return;
  }
  const config = await productConfig(name);
  if (command === "start") {
    await prepareProduct(config);
    await composeProduct(config, [
      "--profile",
      "agent",
      "up",
      "-d",
      "--build",
      "--wait",
      "--wait-timeout",
      "180",
    ]);
    console.log("Local dashboard: http://127.0.0.1:" + config.port);
  } else if (command === "stop")
    await composeProduct(config, ["--profile", "agent", "down"]);
  else if (command === "status") await composeProduct(config, ["ps"]);
  else if (command === "reset") {
    if (extra[0] !== "--confirm=" + config.id)
      throw Error(
        "Reset destroys only this installation data. Require --confirm=" +
          config.id,
      );
    await composeProduct(config, ["--profile", "agent", "down", "--volumes"]);
    console.log("Removed installation volumes; operator credentials retained.");
  } else if (command === "credentials")
    console.log(join(config.dir, "operators.json"));
  else
    throw Error(
      "Use init/start/stop/status/credentials/reset [installation-name]",
    );
}

FILE scripts/source-manifest.mjs
import { readdir, readFile } from "node:fs/promises";
import { createHash } from "node:crypto";
import { join, relative } from "node:path";
import { root } from "./opa-runtime.mjs";
export async function sourceManifest() {
  const files = [];
  const hash = (value) => createHash("sha256").update(value).digest("hex");
  async function walk(path) {
    const entries = await readdir(path, { withFileTypes: true }).catch(
      () => null,
    );
    if (entries) {
      for (const entry of entries)
        if (
          ![
            "node_modules",
            ".next",
            "test-results",
            "playwright-report",
          ].includes(entry.name) &&
          !entry.name.endsWith(".tsbuildinfo")
        )
          await walk(join(path, entry.name));
    } else
      files.push({
        path: relative(root, path).replaceAll("\\", "/"),
        sha256: hash(await readFile(path)),
      });
  }
  for (const location of [
    "server",
    "src",
    "apps/dashboard",
    "services",
    "infra",
    "scripts",
    "tests",
    "docs/specification",
    "package.json",
    "package-lock.json",
    "tsconfig.json",
    "vite.config.ts",
    "playwright.config.ts",
    ".dockerignore",
  ])
    await walk(join(root, location));
  files.sort((a, b) => a.path.localeCompare(b.path));
  return { sourceHash: hash(JSON.stringify(files)), files };
}

  // 0.0.0.0), which can differ from the browser's loopback Host. Validate the
  // actual Host first and derive the browser origin from that trusted header.

    if (!Number.isSafeInteger(size) || size > 8000)
      throw new Error("Fixture file exceeds read limit");
    const content = await this.git(["cat-file", "blob", entry.sha]);
    if (content.includes("\0"))
      throw new Error("Unsupported binary fixture content");
    return content;
  }

  private async proposal(args: Proposal) {
    await this.checkBase(args.baseSha);
    const entries = await this.entries(args.baseSha);
    const changes = args.changes
      .map((change) => ({ ...change }))
      .sort((a, b) => (a.path < b.path ? -1 : a.path > b.path ? 1 : 0));
    const diffs: string[] = [];
    for (const change of changes) {
      const existing = entries.find(({ path }) => path === change.path);
      if (
        entries.some(
          ({ path }) =>
            path !== change.path &&
            (path.toLowerCase() === change.path.toLowerCase() ||
              path.startsWith(change.path + "/") ||
              change.path.startsWith(path + "/")),
        )
      )
        throw new Error("Fixture path conflicts with existing tree");
      const previous = existing ? await this.content(existing) : "";
      if (existing && previous === change.content) continue;
      if (!existing && change.content === "") {
        // Git represents an empty added file with metadata, never a zero-line hunk.
        diffs.push(
          `diff --git a/${change.path} b/${change.path}\nnew file mode 100644\nindex 0000000..e69de29\n`,
        );
        continue;
      }
      const lines = (text: string, prefix: string) => {
        if (!text) return [];
        const parts = text.split("\n");
        if (parts.at(-1) === "") parts.pop();
        const result = parts.map((line) => prefix + line);
        if (!text.endsWith("\n")) result.push("\\ No newline at end of file");
        return result;
      };
      const count = (text: string) =>
        text ? text.split("\n").length - Number(text.endsWith("\n")) : 0;
      const oldCount = count(previous),
        newCount = count(change.content);
      diffs.push(
        [
          `diff --git a/${change.path} b/${change.path}`,
          ...(existing ? [] : ["new file mode 100644"]),
          `--- ${existing ? `a/${change.path}` : "/dev/null"}`,
          `+++ b/${change.path}`,
          `@@ -${oldCount ? 1 : 0},${oldCount} +${newCount ? 1 : 0},${newCount} @@`,
          ...lines(previous, "-"),
          ...lines(change.content, "+"),
          "",
        ].join("\n"),
      );
    }
    if (diffs.length === 0) throw new Error("Proposal has no changes");
    const canonical = {
      repositoryId: REPOSITORY.id,
      owner: REPOSITORY.owner,
      repo: REPOSITORY.fullName,
      baseSha: args.baseSha,
      branch: args.branch,
      changes,
    };
    return {
      repo: REPOSITORY.fullName,
      baseSha: args.baseSha,
      branch: args.branch,
      changes,
      diff: diffs.join(""),
      proposalFingerprint: createHash("sha256")
        .update(JSON.stringify(canonical))
        .digest("hex"),
      sideEffects: false,
    };
  }

  private async writeTree(entries: TreeEntry[]): Promise<string> {
    const leaves: string[] = [];
    const children = new Map<string, TreeEntry[]>();
    for (const entry of entries) {
      const slash = entry.path.indexOf("/");
      if (slash < 0)
        leaves.push(
          `${entry.mode} ${entry.type} ${entry.sha}\t${entry.path}\0`,
        );
      else {
        const directory = entry.path.slice(0, slash);
        const group = children.get(directory) ?? [];
        group.push({ ...entry, path: entry.path.slice(slash + 1) });
        children.set(directory, group);
      }
    }
    for (const [directory, group] of children)
      leaves.push(`040000 tree ${await this.writeTree(group)}\t${directory}\0`);
    return (await this.git(["mktree", "-z"], leaves.join(""))).trim();
  }

  private git(args: string[], input = ""): Promise<string> {
    const nullFile = process.platform === "win32" ? "NUL" : "/dev/null";
    // Inherit only OS executable lookup/temp paths, never credentials, Git config,
    // object directories, command injection variables or the user's home config.
    const env = Object.fromEntries(
      Object.entries(process.env).filter(([key]) =>
        ["PATH", "SYSTEMROOT", "WINDIR", "TEMP", "TMP", "PATHEXT"].includes(
          key.toUpperCase(),
        ),
      ),
    );
    return new Promise((resolveResult, reject) => {
      const child = spawn(
        "git",
        [
          "-c",
          `core.hooksPath=${nullFile}`,
          "-c",
          "commit.gpgsign=false",
          "-c",
          "tag.gpgsign=false",
          "-c",
          "core.fsmonitor=false",
          "-c",
          "gc.auto=0",
          "-c",
          "protocol.allow=never",
          ...args,
        ],
        {
          cwd: this.directory,
          env: {
            ...env,
            GIT_CONFIG_NOSYSTEM: "1",
            GIT_CONFIG_GLOBAL: nullFile,
            GIT_CONFIG_SYSTEM: nullFile,
            GIT_TERMINAL_PROMPT: "0",
            GIT_ALLOW_PROTOCOL: "",
            LC_ALL: "C",
          },
          shell: false,
          windowsHide: true,
          stdio: [input.length ? "pipe" : "ignore", "pipe", "pipe"],
        },
      );
      const chunks: Buffer[] = [];
      let bytes = 0,
        failed = false,
        stdinErrorCode: string | undefined,
        timedOut = false;
      const fail = () => {
        failed = true;
        child.kill();
      };
      const timer = setTimeout(() => {
        timedOut = true;
        fail();
      }, 5000);
      child.stdout!.on("data", (chunk: Buffer) => {
        bytes += chunk.length;
        if (bytes > 256000) fail();
        else chunks.push(chunk);
      });
      child.stderr!.on("data", (chunk: Buffer) => {
        bytes += chunk.length;
        if (bytes > 256000) fail();
      });
      child.stdin?.on("error", (error: NodeJS.ErrnoException) => {
        stdinErrorCode = error.code;
        fail();
      });
      child.on("error", () => {
        clearTimeout(timer);
        reject(new Error("Git fixture process unavailable"));
      });
      child.on("close", (code) => {
        clearTimeout(timer);
        if (failed || code !== 0)
          reject(
            new Error(
              "Git fixture operation failed; reconcile before retrying",
              {
                cause: {
                  operation: args[0],
                  exitCode: code,
                  stdinErrorCode,
                  timedOut,
                  outputLimit: bytes > 256000,
                },
              },
            ),
          );
        else resolveResult(Buffer.concat(chunks).toString("utf8"));
      });
      child.stdin?.end(input);
    });
  }
}

    parsed = new URL(url);

      !["postgres:", "postgresql:"].includes(parsed.protocol) ||

      !parsed.password ||

      throw new Error();

      port: parsed.port ? Number(parsed.port) : 5432,

      password: decodeURIComponent(parsed.password),
      ssl: false,
      max: 2,
      connectionTimeoutMillis: 2000,
      statement_timeout: 1500,
      lock_timeout: 1000,
      idle_in_transaction_session_timeout: 5000,
    };
  } catch {
    // Never expose a connection string (including credentials) in an error.
    throw new Error("Invalid local PostgreSQL fixture configuration");
  }

const roleSafety = `SELECT
  current_user = $1 AND current_database() = 'agentbrake_target_fixture'
  AND NOT (r.rolsuper OR r.rolcreatedb OR r.rolcreaterole OR r.rolbypassrls OR r.rolreplication)
  AND NOT EXISTS (SELECT 1 FROM pg_catalog.pg_auth_members WHERE member = r.oid)
  AND NOT EXISTS (SELECT 1 FROM pg_catalog.pg_database WHERE datdba = r.oid)

export class PostgresFixtureAdapter {
  private reader: pg.Pool;
  private writer: pg.Pool;
  private closed = false;

  constructor(options: { readUrl: string; writeUrl: string }) {
    const read = connection(options.readUrl, "fixture_reader");
    const write = connection(options.writeUrl, "fixture_writer");
    if (read.host !== write.host || read.port !== write.port)
      throw new Error("Invalid local PostgreSQL fixture configuration");
    this.reader = new pg.Pool(read);
    this.writer = new pg.Pool(write);
    // Idle connection errors must not crash the gateway or log credentials.
    this.reader.on("error", () => {});
    this.writer.on("error", () => {});
  }

  async execute(action: ExecutionAction): Promise<unknown> {
    if (this.closed) throw new Error("PostgreSQL fixture adapter is closed");
    if (!action || action.sandbox !== true)
      throw new Error("PostgreSQL fixture requires sandbox execution");
    if (typeof action.tool !== "string" || !Object.hasOwn(schemas, action.tool))
      throw new Error("Unsupported PostgreSQL fixture tool");
    const parsed = schemas[action.tool as keyof typeof schemas].safeParse(
      action.arguments,
    );
    if (!parsed.success)
      throw new Error("Invalid PostgreSQL fixture arguments");
    if (action.tool === "database_schema") {
      // Fixed template metadata: no general catalog browsing or user-controlled SQL.
      return {
        tables: [
          {
            schema: "public",
            name: "agentbrake_customers",
            columns: [
              { name: "id", type: "integer" },
              { name: "name", type: "text" },
              { name: "plan", type: "text" },
              { name: "status", type: "text", values: ["active", "paused"] },
              { name: "version", type: "integer" },
            ],
          },
        ],
        maximumRows: 100,
      };
    }
    const read = action.tool === "database_read";
    const expected = read
      ? undefined
      : (action.precondition?.version ??
        expectedVersion(await this.preview(action)));
    if (
      !read &&
      (typeof expected !== "number" ||
        !Number.isSafeInteger(expected) ||
        expected < 0)
    )
      throw Error("Invalid target precondition");
    const pool = read ? this.reader : this.writer;
    let client: pg.PoolClient | undefined;
    try {
      client = await pool.connect();
      await client.query(read ? "BEGIN READ ONLY" : "BEGIN READ WRITE");
      const safety = await client.query(roleSafety, [
        read ? "fixture_reader" : "fixture_writer",
      ]);
      if (safety.rows.length !== 1 || safety.rows[0].safe !== true)
        throw new Error("Unsafe PostgreSQL fixture role");
      let output: unknown;
      if (read) {
        const { limit } = parsed.data as { limit: number };
        const result = await client.query(
          "SELECT id,name,plan,status,version FROM public.agentbrake_customers ORDER BY id LIMIT $1",
          [limit],
        );
        output = { rows: result.rows };
      } else {
        const { id, status } = parsed.data as {
          id: number;
          status: "active" | "paused";
        };
        const result = await client.query(
          "SELECT id,status,version FROM public.agentbrake_update_customer_checked($1::integer,$2::text,$3::integer)",
          [id, status, expected],
        );
        if (result.rows.length === 0) throw new TargetChangedError();
        if (result.rows.length !== 1)
          throw new Error("Unexpected PostgreSQL fixture row count");
        output = result.rows[0];
      }
      await client.query("COMMIT");
      return output;
    } catch (error) {
      if (client) await client.query("ROLLBACK").catch(() => {});
      if (error instanceof TargetChangedError) throw error;
      if (
        error instanceof Error &&
        ["Record not found", "Unsafe PostgreSQL fixture role"].includes(
          error.message,
        )
      )
        throw error;
      throw new Error("PostgreSQL fixture execution failed");
    } finally {
      client?.release();
    }
  }

  async preview(action: ExecutionAction): Promise<ActionPreview> {
    if (action.sandbox !== true || !Object.hasOwn(schemas, action.tool))
      throw Error("Unsupported database preview");
    if (action.tool !== "database_update")
      return {
        kind: "operation",
        summary: "Exact bounded database read template",
        details: {
          tool: action.tool,
          arguments: schemas[action.tool as keyof typeof schemas].parse(
            action.arguments,
          ),
        },
      };
    const args = schemas.database_update.parse(action.arguments);

      tool: "database_read",
      arguments: { limit: 100 },
      sandbox: true,
    })) as { rows: { id: number; status: string; version: number }[] };
    const row = result.rows.find((row) => row.id === args.id);
    if (!row) throw Error("Record not found");
    return databasePreview(row, args.status);
  }

  async close() {
    if (this.closed) return;
    this.closed = true;
    await Promise.all([this.reader.end(), this.writer.end()]);
  }
}

      version: before.version + 1,
    };

    ? rowSchema.parse(preview.details.before).version

import { createHmac } from "node:crypto";
import { Store } from "./store.js";
import type { ControlStore } from "./control-store.js";
import { TargetChangedError, type ActionPreview } from "./contracts.js";
import { databasePreview, expectedVersion } from "./adapters/preview.js";
import {
  FixtureHttpAdapter,
  type ExecutionAction,
} from "./adapters/fixture-http.js";
export type ConnectionConfig = {
  githubToken?: string;
  githubRepository?: string;
  databaseUrl?: string;
  fixture?: { url: string; token: string };
};
export class Connectors {
  private fixture?: FixtureHttpAdapter;
  constructor(
    private store: ControlStore,
    public config: ConnectionConfig,
    private bindingKey: string,
  ) {
    if (config.fixture)
      this.fixture = new FixtureHttpAdapter(
        config.fixture.url,
        config.fixture.token,
      );
  }
  binding(tool: string, sandbox: boolean) {
    const destination = sandbox
      ? this.config.fixture
        ? JSON.stringify(this.config.fixture)
        : "sandbox"
      : tool.startsWith("database_")
        ? this.config.databaseUrl || "sandbox"
        : JSON.stringify([
            this.config.githubRepository,
            this.config.githubToken,
          ]);
    return createHmac("sha256", this.bindingKey)
      .update(destination)
      .digest("hex");
  }
  modes() {
    return {
      github:
        this.config.githubToken && this.config.githubRepository
          ? "configured"
          : "sandbox",
      database: this.config.databaseUrl ? "configured" : "sandbox",
      repository: this.config.githubRepository || "agentbrake/sandbox",
    };
  }
  async preview(a: ExecutionAction): Promise<ActionPreview> {
    if (!a.sandbox) throw Error("Only local fixture previews are available");
    if (this.fixture) return this.fixture.preview(a);
    if (!(this.store instanceof Store))
      throw Error("Target preview unavailable");
    if (a.tool === "database_update") {
      const row = this.store.db
        .prepare("SELECT id,status,version FROM customers WHERE id=?")
        .get(a.arguments.id as number);
      if (!row) throw Error("Record not found");
      return databasePreview(row, a.arguments.status);
    }
    return {
      kind: "operation",
      summary: "Exact local fixture operation",
      details: { tool: a.tool, arguments: a.arguments },
    };
  }
  async execute(a: ExecutionAction): Promise<unknown> {
    if (!a.sandbox)
      throw new Error(
        "Real service execution is disabled. A separately approved sandbox connector is required.",
      );
    if (this.fixture) return this.fixture.execute(a);
    if (!(this.store instanceof Store))
      throw new Error(
        "PostgreSQL control store requires explicit isolated target transport",
      );
    const args = a.arguments;
    if (a.tool === "database_read") {
      return {
        rows: this.store.db
          .prepare("SELECT * FROM customers ORDER BY id LIMIT ?")
          .all(args.limit as number),
      };
    }
    if (a.tool === "database_update") {
      const version = a.precondition?.version ?? expectedVersion(a.preview);
      const row = this.store.db
        .prepare(
          "UPDATE customers SET status=?,version=version+1 WHERE id=? AND (? IS NULL OR version=?) RETURNING id,status,version",
        )
        .get(
          args.status as string,
          args.id as number,
          version ?? null,
          version ?? null,
        );
      if (!row) {
        if (version !== undefined) throw new TargetChangedError();
        throw new Error("Record not found");
      }
      return row;
    }
    if (a.tool === "github_read_file" || a.tool === "github_create_issue") {
      if (a.sandbox || !this.config.githubToken) {
        if (a.tool === "github_read_file")
          return {
            path: args.path,
            content:
              "# AgentBrake sandbox\nA local repository fixture. No GitHub request was sent.",
            sandbox: true,
          };
        const row = this.store.db
          .prepare(
            "INSERT INTO sandbox_issues(title,body) VALUES (?,?) RETURNING id",
          )
          .get(args.title as string, args.body as string)!;
        return { number: row.id, title: args.title, sandbox: true };
      }
    }
    throw new Error("Unsupported tool");
  }
  async close() {
    // Local in-memory fixture has no outbound session.
  }
}

export function redactExport(value: unknown): unknown {
  if (typeof value === "string") return value.replace(credential, "[REDACTED]");
  if (Array.isArray(value)) return value.map(redactExport);
  if (value && typeof value === "object")
    return Object.fromEntries(

          key,

}

  });

      error: unknown,
      _req: express.Request,

        error instanceof HttpError
          ? error.status
          : error instanceof z.ZodError || error instanceof SyntaxError
            ? 400
            : (error as { status?: number })?.status === 413
              ? 413
              : 503;
      res

          error:
            error instanceof HttpError
              ? error.message
              : "Private Gateway request failed safely",

export class RemoteGateway implements GatewayPort {
  private url: URL;
  constructor(
    url: string,
    private key: string,
  ) {
    this.url = new URL(url);
    if (
      this.url.protocol !== "http:" ||
      !["gateway-control", "127.0.0.1", "localhost"].includes(
        this.url.hostname,
      ) ||
      this.url.username ||
      this.url.password ||
      this.url.search ||
      this.url.hash ||
      key.length < 32
    )
      throw Error("Only private local Gateway bridge is permitted");
  }
  private async call(path: string, body: unknown) {
    let response: Response;
    try {
      response = await fetch(new URL(path, this.url), {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: "Bearer " + this.key,
        },
        body: JSON.stringify(body),
        signal: AbortSignal.timeout(10000),
        redirect: "error",
      });
    } catch {
      throw new HttpError(
        503,
        "Gateway response unavailable. Refresh the action state before any further operation.",
      );
    }
    const reader = response.body!.getReader();
    let size = 0;
    const chunks: Uint8Array[] = [];
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      size += value.length;
      if (size > 256000) {
        await reader.cancel();
        throw new HttpError(503, "Gateway response exceeded the bound");
      }
      chunks.push(value);
    }
    const result = JSON.parse(Buffer.concat(chunks).toString());
    if (!response.ok)
      throw new HttpError(
        [400, 401, 403, 404, 409, 413, 429].includes(response.status)
          ? response.status
          : 503,
        typeof result.error === "string"
          ? result.error
          : "Gateway request failed",
      );
    return result;
  }
  submit(agentId: string, input: unknown) {
    return this.call("/submit", { agentId, input });
  }
  review(id: string) {
    return this.call("/review", { id });
  }
  approve(id: string, approve: boolean, reviewer?: ReviewContext) {
    return this.call("/approve", { id, approve, reviewer });
  }
  async actionForAgent(_agentId: string, _id: string): Promise<never> {
    throw new HttpError(
      404,
      "Agent actions require the agent Gateway endpoint",
    );
  }
}

    const request = actionSchema.safeParse(input);
    if (!request.success) throw new HttpError(400, "Invalid action envelope");
    const fingerprint = digest(canonical(request.data));
    const key = agentId + ":" + request.data.requestId;
    const running = this.inflight.get(key);
    if (running) {
      if (running.digest !== fingerprint)
        throw new HttpError(
          409,
          "Request ID is already bound to different arguments",
        );
      return agentRecord(await running.result);
    }
    const release = this.limits.enter(agentId);
    const result = this.submitOnce(agentId, request.data);
    this.inflight.set(key, { digest: fingerprint, result });
    try {
      return agentRecord(await result);
    } finally {
      this.inflight.delete(key);
      release();
    }
  }
  private taskFingerprint(agent: Agent) {
    const { used_calls, ...grant } = agent.grant ?? {};
    return digest(canonical(grant));
  }
  async actionForAgent(agentId: string, id: string) {
    const action = await this.store.action(id),
      agent = await this.store.agent(agentId);
    if (!action || !agent || action.agentId !== agentId)
      throw new HttpError(404, "Action not found");
    // Status recovery does not consume another execution quota. Revoke/expiry
    // and task changes still withdraw access to any cached protected result.
    const control = await this.store.settings();
    const grant = grantSchema.safeParse(agent.grant);
    if (
      !grant.success ||
      grant.data.revoked ||
      grant.data.expires_at_ms <= Date.now() ||
      agent.status !== "active" ||
      control.paused ||
      action.policyRevision !== control.revision ||
      action.policyHash !== policyHash() ||
      action.grantFingerprint !== this.taskFingerprint(agent)
    )
      throw new HttpError(
        403,
        "Current grant no longer permits access to this action",
      );
    return agentRecord(action);
  }
  private contextFingerprint(agent: Agent, settings: Settings) {
    return digest(
      canonical({
        grant: this.taskFingerprint(agent),
        status: agent.status,
        tokenAudience: agent.tokenAudience,
        tokenExpiresAt: agent.tokenExpiresAt,
        settings,
        tools: agent.tools,
        resources: agent.resources,
        sandbox: agent.sandbox,
      }),
    );
  }
  private async authorize(
    agent: Agent,
    tool: string,
    args: Record<string, unknown>,
  ) {
    const settings = await this.store.settings();
    const decision = this.evaluateWithSettings(agent, tool, args, settings);
    const decisionId = randomUUID();
    const context = this.contextFingerprint(agent, settings);
    const revision = settings.revision;
    let policy: PolicyDecision = {
      effect: "DENY",
      reason_code: decision.rule,
      decision_id: decisionId,
      policy_version: POLICY_VERSION,
      policy_hash: policyHash(),
    };
    if (decision.status !== "blocked") {
      const input: PolicyInput = {
        decision_id: decisionId,
        now_ms: Date.now(),
        identity: {
          agent_id: agent.id,
          tenant: "local",
          audience: AUDIENCE,
          expires_at_ms: agent.tokenExpiresAt!,
        },
        grant: agent.grant!,
        action: {
          tool,
          rows:
            tool === "database_read"
              ? Number(decision.args.limit)
              : tool === "database_update"
                ? 1
                : 0,
        },
        target: { id: decision.resource, environment: "sandbox" },
        controls: {
          paused: settings.paused,
          agent_active: agent.status === "active",
          tainted: agent.tainted,
          ...settings.policies,
        },
      };
      policy = await this.policy.evaluate(input);
      if (policy.effect === "DENY") {
        decision.status = "blocked";
        decision.rule = policy.reason_code;
        decision.reason = "The policy engine denied this action.";
        decision.args = {};
      } else if (policy.effect === "REQUIRE_APPROVAL")
        decision.status = "pending";
    }
    const current = await this.store.agent(agent.id);
    if (
      decision.status !== "blocked" &&
      (!current ||
        this.contextFingerprint(current, await this.store.settings()) !==
          context ||
        (await this.evaluate(current, tool, args)).status === "blocked")
    ) {
      decision.status = "blocked";
      decision.rule = "authority-changed";
      decision.reason = "Authority changed while awaiting the policy engine.";
      decision.args = {};
    }
    return { decision, policy, context, revision };
  }
  private async submitOnce(agentId: string, input: unknown) {
    const parsed = actionSchema.safeParse(input);
    if (!parsed.success) throw new HttpError(400, "Invalid action envelope");
    const request = parsed.data;
    const agent = await this.store.agent(agentId);
    if (!agent) throw new HttpError(401, "Unknown agent");
    const fingerprint = digest(canonical(request));
    const previous = await this.store.previous(agentId, request.requestId);
    if (previous) {
      if (
        (await this.evaluate(agent, request.tool, request.arguments)).status ===
        "blocked"
      )
        throw new HttpError(
          403,
          "Current authority no longer permits access to this action",
        );
      if (previous.digest !== fingerprint)
        throw new HttpError(
          409,
          "Request ID is already bound to different arguments",
        );
      return this.actionForAgent(agentId, previous.id);
    }
    const started = performance.now();
    const { decision, policy, context, revision } = await this.authorize(
      agent,
      request.tool,
      request.arguments,
    );
    const record: ActionRecord = {
      id: randomUUID(),
      agentId,
      requestId: request.requestId,
      tool:
        Object.hasOwn(toolSchemas, request.tool) ||
        ["database_delete", "http_request"].includes(request.tool)
          ? request.tool
          : "unregistered",
      digest: fingerprint,
      destination: "",
      correlationId: randomUUID(),
      grantId: agent.grant?.id,
      grantFingerprint: this.taskFingerprint(agent),
      policyRevision: revision,
      policyDecisionId: policy.decision_id,
      policyVersion: policy.policy_version,
      policyHash: policy.policy_hash,
      policyEffect: policy.effect,
      arguments: decision.args,
      status: decision.status,
      rule: decision.rule,
      reason: decision.reason,
      resource: decision.resource,
      sandbox:
        agent.sandbox ||
        (request.tool.startsWith("database_")
          ? this.connectors.modes().database === "sandbox"
          : this.connectors.modes().github === "sandbox"),
      createdAt: new Date().toISOString(),
      expiresAt: new Date(Date.now() + 15 * 60_000).toISOString(),
      durationMs: Math.round((performance.now() - started) * 100) / 100,
    };
    // Passing policy is not execution success. This state also survives an audit failure or process crash.
    record.destination = this.connectors.binding(record.tool, record.sandbox);
    if (record.status === "pending") {
      try {
        record.preview = actionPreviewSchema.parse(
          await this.connectors.preview(record),
        );
        if (secretPattern.test(JSON.stringify(record.preview)))
          throw Error("Sensitive preview");
        record.review = issueReview(record);
      } catch {
        record.status = "blocked";
        record.rule = "preview-unavailable";
        record.reason =
          "A safe, current preview could not be obtained. No mutation was attempted.";
        delete record.preview;
      }
    }
    if (record.status === "allowed") record.status = "queued";
    await this.store.insertAction(record);
    await this.store.audit("action." + record.status, {
      actionId: record.id,
      agentId,
      tool: record.tool,
      rule: record.rule,
      correlationId: record.correlationId,
      grantId: record.grantId,
      decisionId: record.policyDecisionId,
      policyHash: record.policyHash,
      policyRevision: record.policyRevision,
      effect: record.policyEffect,
    });
    this.notify();
    if (record.status === "queued") return this.execute(record, context);
    return record;
  }
  async review(id: string) {
    const action = await this.store.action(id);
    if (!action) throw new HttpError(404, "Action not found");
    if (
      action.status !== "pending" ||
      !action.review ||
      !matchesReview(action, action.review)
    )
      throw new HttpError(409, "Action requires a new request and review");
    return { action, proof: action.review };
  }
  async approve(id: string, approve: boolean, reviewer?: ReviewContext) {
    const record = await this.store.action(id);
    if (!record) throw new HttpError(404, "Action not found");
    if (record.status !== "pending")
      throw new HttpError(409, "This action is no longer awaiting approval");
    const owner = await this.store.agent(record.agentId);
    if (reviewer) {
      if (
        reviewer.actor.tenant !== "local" ||
        !["administrator", "approver"].includes(reviewer.actor.role)
      )
        throw new HttpError(403, "Approver role required");
      if (
        approve &&
        reviewer.team &&
        reviewer.actor.id === owner?.grant?.human_owner &&
        (githubWrites.has(record.tool) || record.tool === "database_update")
      )
        throw new HttpError(403, "A different human must review this change");
      if (
        (reviewer.team || reviewer.proof) &&
        (!reviewer.proof || !matchesReview(record, reviewer.proof))
      )
        throw new HttpError(
          409,
          "Reviewed action fingerprint or nonce changed",
        );
    }
    record.approvalReceipt = {
      actorId: reviewer?.actor.id ?? "local-administrator",
      tenant: "local",
      decision: approve ? "approve" : "reject",
      at: new Date().toISOString(),
      proof: record.review ?? issueReview(record),
    };
    if (
      record.policyHash !== policyHash() ||
      record.policyRevision !== (await this.store.settings()).revision ||
      record.grantFingerprint !==
        this.taskFingerprint((await this.store.agent(record.agentId))!)
    ) {
      record.status = "blocked";
      record.rule = "approval-context-changed";
      record.reason =
        "Task grant or policy revision changed. A new request and review are required.";
      await this.store.blockPending(record);
      this.notify();
      return record;
    }
    if (!approve) {
      record.status = "blocked";
      record.reason = "An administrator rejected this action.";
      record.rule = "human-rejected";
    } else if (Date.parse(record.expiresAt) <= Date.now()) {
      record.status = "blocked";
      record.reason = "The 15-minute approval window expired.";
      record.rule = "approval-expired";
    } else if (
      record.destination !==
      this.connectors.binding(record.tool, record.sandbox)
    ) {
      record.status = "blocked";
      record.rule = "destination-changed";
      record.reason =
        "Connection configuration changed after this request. Submit a new action for review.";
    } else {
      let currentPreview;
      try {
        currentPreview = await this.connectors.preview(record);
      } catch {
        currentPreview = undefined;
      }
      if (
        !record.preview ||
        canonical(currentPreview ?? null) !== canonical(record.preview)
      ) {
        record.status = "blocked";
        record.rule = "preview-changed";
        record.reason =
          "The target or preview changed after review. Submit a new action.";
        await this.store.blockPending(record);
        this.notify();
        return record;
      }
      const { decision, policy, context } = await this.authorize(
        (await this.store.agent(record.agentId))!,
        record.tool,
        record.arguments,
      );
      if ((await this.store.action(id))?.status !== "pending")
        throw new HttpError(409, "This action is no longer awaiting approval");
      if (decision.status === "blocked") {
        record.status = "blocked";
        record.reason = decision.reason;
        record.rule = decision.rule;
      } else {
        await this.store.audit("approval.granted", {
          actionId: id,
          actor: record.approvalReceipt.actorId,
          fingerprint: record.approvalReceipt.proof.fingerprint,
        });
        record.policyDecisionId = policy.decision_id;
        return this.execute(record, context);
      }
    }
    await this.store.blockPending(record);
    this.notify();
    return record;
  }
  private async execute(record: ActionRecord, context: string) {
    const reserved = await this.store.reserve(
      record,
      (current, settings, persisted) => {
        if (
          persisted.policyHash !== policyHash() ||
          persisted.policyRevision !== settings.revision ||
          persisted.grantFingerprint !== this.taskFingerprint(current)
        )
          return {
            rule: "approval-context-changed",
            reason:
              "Task grant or policy changed. A new request and review are required.",
          };
        const recheck = this.evaluateWithSettings(
          current,
          record.tool,
          record.arguments,
          settings,
        );
        const expired =
          record.status === "pending" &&
          (!Number.isFinite(Date.parse(record.expiresAt)) ||
            Date.parse(record.expiresAt) <= Date.now());
        const bound = (a: ActionRecord) =>
          canonical({
            agentId: a.agentId,
            requestId: a.requestId,
            tool: a.tool,
            arguments: a.arguments,
            digest: a.digest,
            destination: a.destination,
            grantFingerprint: a.grantFingerprint,
            policyRevision: a.policyRevision,
            policyHash: a.policyHash,
            expiresAt: a.expiresAt,
            sandbox: a.sandbox,
            review: a.review,
            preview: a.preview,
          });
        const swapped =
          bound(record) !== bound(persisted) ||
          record.destination !==
            this.connectors.binding(record.tool, record.sandbox);
        if (
          expired ||
          recheck.status === "blocked" ||
          swapped ||
          this.contextFingerprint(current, settings) !== context
        )
          return {
            rule: expired
              ? "approval-expired"
              : recheck.status === "blocked"
                ? recheck.rule
                : swapped
                  ? "action-changed"
                  : "authority-changed",
            reason:
              "Authority or approved action changed before dispatch. No tool was called.",
          };
        return undefined;
      },
    );
    if (!reserved) {
      this.notify();
      return record;
    }
    this.notify();
    try {
      const result = await this.connectors.execute(record);
      if (secretPattern.test(JSON.stringify(result))) {
        record.status = "blocked";
        record.rule = "secret-output";
        record.reason = "Credential-like tool output was withheld.";
      } else {
        record.status = "allowed";
        record.result = result;
        record.reason = "Executed through the enforced gateway.";
      }
    } catch (error) {
      record.status =
        error instanceof TargetChangedError
          ? "blocked"
          : readTools.has(record.tool)
            ? "failed"
            : "unknown";
      if (error instanceof TargetChangedError) record.rule = "target-changed";
      record.reason =
        error instanceof TargetChangedError
          ? "Reviewed target changed at dispatch; no mutation was applied."
          : record.status === "unknown"
            ? "Destination outcome is uncertain. Verify it before creating a new request."
            : "The connector could not complete the request. No result was released.";
    }
    await this.store.saveAction(record);
    await this.store.audit("execution." + record.status, {
      actionId: record.id,
    });
    this.notify();
    return record;
  }
}

import type { Express, Request } from "express";
import { randomUUID } from "node:crypto";
import { z } from "zod";
import { McpServer, createMcpHandler } from "@modelcontextprotocol/server";
import { toNodeHandler } from "@modelcontextprotocol/node";
import {
  HttpError,
  toolDescriptions,
  toolSchemas,
  type Tool,
} from "./contracts.js";
import type { ControlStore } from "./control-store.js";
import type {GatewayPort} from './gateway-port.js';
import { assertPayloadLimits, RequestLimits } from "./security/limits.js";

export const MCP_VERSIONS = ["2026-07-28", "2025-11-25"] as const;
const envelope = z
  .object({
    jsonrpc: z.literal("2.0"),
    id: z.union([z.string().max(128), z.number()]).optional(),
    method: z.string().max(100),
    params: z.record(z.string(), z.unknown()).optional(),
  })
  .strict();
const methods = new Set([
  "server/discover",
  "initialize",
  "notifications/initialized",
  "ping",
  "tools/list",
  "tools/call",
]);

// Authentication is outside MCP metadata. A fresh server is created per request;
// no protocol session can carry authority from one principal to another.
export function mountMcp(app: Express, store: ControlStore, gateway: GatewayPort) {
  const limits = new RequestLimits();
  const handler = createMcpHandler(
    async ({ authInfo }) => {
      const agent = (await store.agent(authInfo!.clientId))!;
      const server = new McpServer({ name: "AgentBrake", version: "0.2.0" });
      for (const name of Object.keys(toolSchemas) as Tool[]) {
        if (!agent.grant?.actions.includes(name) || !agent.tools.includes(name))
          continue;
        server.registerTool(
          name,
          {
            description: toolDescriptions[name],
            inputSchema: toolSchemas[name].extend({
              requestId: z.string().uuid().optional(),
            }),
          },
          async (input: Record<string, unknown> & { requestId?: string }) => {
            const { requestId, ...args } = input;
            try {
              const action = await gateway.submit(agent.id, {
                requestId: requestId ?? randomUUID(),
                tool: name,
                arguments: args,
              });
              return {
                content: [
                  { type: "text" as const, text: JSON.stringify(action) },
                ],
                isError: ["blocked", "failed", "unknown"].includes(
                  action.status,
                ),
              };
            } catch (error) {
              return {
                content: [
                  {
                    type: "text" as const,
                    text: JSON.stringify({
                      error:
                        error instanceof HttpError
                          ? error.message
                          : "Request could not be completed safely",
                    }),
                  },
                ],
                isError: true,
              };
            }
          },
        );
      }
      server.registerTool(
        "action_status",
        {
          description:
            "Read your own action outcome. Pending is not executed. Never retry an uncertain write with a new ID.",
          inputSchema: z.object({ id: z.string().uuid() }).strict(),
        },
        async ({ id }) => {
          const action = await gateway.actionForAgent(agent.id, id);
          return {
            content: [{ type: "text" as const, text: JSON.stringify(action) }],
          };
        },
      );
      return server;
    },
    { legacy: "stateless", responseMode: "json", maxRequestBodySize: 32768 },
  );
  const nodeHandler = toNodeHandler(handler, { maxRequestBodySize: 32768 });
  app.all("/mcp", async (req, res) => {
    const correlationId = randomUUID();
    res.setHeader("X-Correlation-ID", correlationId);
    res.setHeader("Cache-Control", "no-store");
    let release: (() => void) | undefined;
    try {
      const bearer = req.headers.authorization;
      const agent = bearer?.startsWith("Bearer ")
        ? await store.authenticate(bearer.slice(7))
        : undefined;
      if (
        !agent ||
        agent.status !== "active" ||
        !agent.grant ||
        agent.grant.revoked ||
        agent.grant.expires_at_ms <= Date.now()
      )
        throw new HttpError(
          401,
          "Valid active agent credential and grant required",
        );
      release = limits.enter(agent.id);
      if (req.method !== "POST")
        throw new HttpError(405, "Only stateless POST requests are supported");
      assertPayloadLimits(req.body);
      const parsed = envelope.safeParse(req.body);
      if (!parsed.success) throw new HttpError(400, "Invalid MCP envelope");
      const body = parsed.data;
      const version = req.headers["mcp-protocol-version"];
      if (
        typeof version !== "string" ||
        !MCP_VERSIONS.includes(version as (typeof MCP_VERSIONS)[number])
      )
        throw new HttpError(400, "Unsupported or missing MCP protocol version");
      if (!methods.has(body.method))
        throw new HttpError(400, "Unsupported MCP method");
      if (
        body.method === "initialize" &&
        (version !== "2025-11-25" || body.params?.protocolVersion !== version)
      )
        throw new HttpError(400, "Initialization protocol version mismatch");
      if (body.method === "tools/call" && body.id === undefined)
        throw new HttpError(400, "Tool calls must have a request ID");
      // Do not let SDK unknown-tool errors echo arbitrary, possibly secret,
      // identifiers. Only this trusted catalog can be named in diagnostics.
      if (
        body.method === "tools/call" &&
        (typeof body.params?.name !== "string" ||
          (!Object.hasOwn(toolSchemas, body.params.name) &&
            body.params.name !== "action_status"))
      )
        throw new HttpError(400, "Tool is not registered");
      // The SDK checks modern header/body agreement and required envelope keys.
      // Reject identity-bearing metadata; it is never a task grant.
      const meta = body.params?._meta;
      if (
        meta &&
        typeof meta === "object" &&
        Object.keys(meta).some(
          (key) =>
            ![
              "io.modelcontextprotocol/protocolVersion",
              "io.modelcontextprotocol/clientInfo",
              "io.modelcontextprotocol/clientCapabilities",
              "io.modelcontextprotocol/logLevel",
              "traceparent",
              "tracestate",
              "baggage",
            ].includes(key),
        )
      )
        throw new HttpError(400, "Unsupported MCP metadata");
      (req as Request & { auth: unknown }).auth = {
        token: "",
        clientId: agent.id,
        scopes: [],
      };
      await nodeHandler(req, res, body);
      await store.audit("mcp.response", {
        correlationId,
        agentId: agent.id,
        method: body.method,
        protocolVersion: version,
        status: res.statusCode,
      });
    } catch (error) {
      const status = error instanceof HttpError ? error.status : 500;
      try {
        await store.audit("mcp.rejected", { correlationId, status });
      } catch {
        if (!res.headersSent)
          res.status(503).json({ error: "Audit unavailable" });
        return;
      }
      if (!res.headersSent)
        res.status(status).json({
          jsonrpc: "2.0",
          id: null,
          error: {
            code: -32600,
            message:
              error instanceof HttpError
                ? error.message
                : "Request could not be completed safely",
          },
        });
    } finally {
      release?.();
    }
  });
  return () => handler.close();
}

import { readFileSync } from "node:fs";
import { createHash } from "node:crypto";
import {
  POLICY_VERSION,
  policyInputSchema,
  decisionSchema,
  type PolicyInput,
  type PolicyDecision,
  type PolicyEvaluator,
} from "./contracts.js";
export function policyHash() {
  return createHash("sha256")
    .update(
      readFileSync(
        new URL("../../services/policy/agentbrake.rego", import.meta.url),
      ),
    )
    .update(
      readFileSync(
        new URL("../../services/policy/authz.rego", import.meta.url),
      ),
    )
    .digest("hex");
}
export class OpaEvaluator implements PolicyEvaluator {
  constructor(
    private options: { url?: string; token?: string; timeoutMs?: number } = {
      url: process.env.OPA_URL,
      token: process.env.OPA_TOKEN,
    },
  ) {}
  async evaluate(input: PolicyInput): Promise<PolicyDecision> {
    const hash = policyHash();
    const deny = (reason_code: string): PolicyDecision => ({
      effect: "DENY",
      reason_code,
      decision_id: input.decision_id,
      policy_version: POLICY_VERSION,
      policy_hash: hash,
    });
    if (!policyInputSchema.safeParse(input).success)
      return deny("invalid-policy-input");
    if (!this.options.url || !this.options.token)
      return deny("pdp-unavailable");
    try {
      const url = new URL("/v1/data/agentbrake/decision", this.options.url);
      if (
        url.protocol !== "http:" ||
        !["127.0.0.1", "localhost", "opa"].includes(url.hostname) ||
        url.username ||
        url.password
      )
        return deny("pdp-invalid-configuration");
      const response = await fetch(url, {
        method: "POST",
        redirect: "error",
        signal: AbortSignal.timeout(this.options.timeoutMs ?? 250),
        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${this.options.token}`,
        },
        body: JSON.stringify({ input }),
      });
      if (
        !response.ok ||
        !response.headers.get("content-type")?.includes("application/json")
      )
        return deny("pdp-error");
      const reader = response.body!.getReader();
      const chunks: Uint8Array[] = [];
      let size = 0;
      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        size += value.length;
        if (size > 16384) {
          await reader.cancel();
          return deny("pdp-invalid-response");
        }
        chunks.push(value);
      }
      const body = JSON.parse(Buffer.concat(chunks).toString("utf8"));
      const result = decisionSchema.safeParse(body.result);
      if (
        !result.success ||
        result.data.decision_id !== input.decision_id ||
        result.data.policy_hash !== hash
      )
        return deny("pdp-invalid-response");
      return result.data;
    } catch (error) {
      return deny(
        error instanceof Error &&
          ["TimeoutError", "AbortError"].includes(error.name)
          ? "pdp-timeout"
          : "pdp-unavailable",
      );
    }
  }
}

  async reserve(
    action: ActionRecord,
    check: ReservationCheck,
  ): Promise<boolean> {
    const result = await this.transaction(async (client) => {
      // All reservation paths use this order. The synchronous caller check runs
      // with the latest authority and persisted envelope while rows stay locked.
      const settings = await this.readSettings(client, true);
      const agents = await client.query(
        "SELECT data FROM agents WHERE id=$1 FOR UPDATE",
        [action.agentId],
      );
      const actions = await client.query(
        "SELECT status,data FROM actions WHERE id=$1 FOR UPDATE",
        [action.id],
      );
      if (!agents.rows[0] || !actions.rows[0])
        throw new HttpError(409, "Action or identity no longer exists");
      const current: Agent = JSON.parse(agents.rows[0].data);
      const persisted: ActionRecord = JSON.parse(actions.rows[0].data);
      if (
        !["pending", "queued"].includes(action.status) ||
        actions.rows[0].status !== action.status ||
        persisted.status !== action.status ||
        persisted.agentId !== current.id ||
        persisted.requestId !== action.requestId
      )
        throw new HttpError(
          409,
          "Execution has already been claimed or its identity changed",
        );
      const denied = check(current, settings, persisted);
      // Start from the persisted envelope; only a newly issued decision ID is
      // carried from the caller after an approval re-evaluation.
      const next: ActionRecord = {
        ...persisted,
        policyDecisionId: action.policyDecisionId,
        approvalReceipt: action.approvalReceipt,
      };
      if (denied) {
        next.status = "blocked";
        next.rule = denied.rule;
        next.reason = denied.reason;
        await this.writeAction(client, next);
        await this.appendAudit(client, "execution.blocked", {
          actionId: next.id,
          rule: next.rule,
        });
        return { allowed: false, next };
      }
      if (!current.grant)
        throw new Error("Reservation requires a current task grant");
      next.status = "executing";
      current.grant.used_calls++;
      if (next.tool === "database_read") current.tainted = true;
      await this.writeAction(client, next);
      await client.query("UPDATE agents SET data=$1 WHERE id=$2", [
        JSON.stringify(current),
        current.id,
      ]);
      await this.appendAudit(client, "execution.started", {
        actionId: next.id,
        approverId: next.approvalReceipt?.actorId,
      });
      return { allowed: true, next };
    });
    // An audit or commit failure must not leave the caller with apparent success.
    Object.assign(action, result.next);
    return result.allowed;
  }

  async blockPending(action: ActionRecord): Promise<void> {
    const next = await this.transaction(async (client) => {
      const { rows } = await client.query(
        "SELECT data,status FROM actions WHERE id=$1 FOR UPDATE",
        [action.id],
      );
      if (!rows[0] || rows[0].status !== "pending")
        throw new HttpError(409, "This action is no longer awaiting approval");
      const persisted: ActionRecord = JSON.parse(rows[0].data);
      const blocked: ActionRecord = {
        ...persisted,
        status: "blocked",
        rule: action.rule,
        reason: action.reason,
        approvalReceipt: action.approvalReceipt,
      };
      await this.writeAction(client, blocked);
      await this.appendAudit(client, "approval.blocked", {
        actionId: action.id,
        rule: action.rule,
      });
      return blocked;
    });
    Object.assign(action, next);
  }
  private async appendAudit(
    client: PoolClient,
    type: string,
    details: unknown,
  ): Promise<void> {
    await client.query("SELECT pg_advisory_xact_lock($1)", [AUDIT_LOCK]);
    const { rows } = await client.query(
      "SELECT hash FROM audit ORDER BY seq DESC LIMIT 1",
    );
    const previous = rows[0]?.hash ?? "genesis";
    const data = JSON.stringify({
      type,
      details,
      at: new Date().toISOString(),
    });
    await client.query(
      "INSERT INTO audit(data,previous_hash,hash) VALUES ($1,$2,$3)",
      [data, previous, hash(previous + data)],
    );
  }

  async audit(type: string, details: unknown): Promise<void> {
    await this.transaction((client) => this.appendAudit(client, type, details));
  }

  async auditTrail(query?: AuditQuery): Promise<AuditRow[]> {
    const values: (string | number)[] = query?.actionId ? [query.actionId] : [];
    const where = query?.actionId
      ? " WHERE data::jsonb->'details'->>'actionId'=$1"
      : "";
    const bounds = query
      ? ` LIMIT $${values.length + 1} OFFSET $${values.length + 2}`
      : "";
    if (query) values.push(query.limit, query.offset ?? 0);
    const { rows } = await this.pool.query(
      "SELECT seq,data,previous_hash,hash FROM audit" +
        where +
        " ORDER BY seq" +
        bounds,
      values,
    );
    return rows.map((row) => ({ ...row, seq: Number(row.seq) }));
  }

  async verifyAudit(): Promise<boolean> {
    let previous = "genesis";
    for (const row of await this.auditTrail()) {
      if (
        row.previous_hash !== previous ||
        row.hash !== hash(previous + row.data)
      )
        return false;
      previous = row.hash;
    }
    return true;
  }

  private async reconcile(): Promise<void> {
    await this.transaction(async (client) => {
      await this.readSettings(client, true);
      const { rows } = await client.query(
        "SELECT data FROM actions WHERE status IN ('queued','executing') ORDER BY created_order FOR UPDATE",
      );
      for (const row of rows) {
        const action: ActionRecord = JSON.parse(row.data);
        action.status = "unknown";
        action.reason =
          "Server restarted during execution. Reconcile the destination before retrying with a new request ID.";
        await this.writeAction(client, action);
        await this.appendAudit(client, "execution.unknown", {
          actionId: action.id,
        });
      }
    });
  }

  async close(): Promise<void> {
    await this.pool.end();
  }
}

export function assertPayloadLimits(input: unknown) {
  const pending = [{ value: input, depth: 0 }];
  let nodes = 0;
  while (pending.length) {
    const { value, depth } = pending.pop()!;
    if (depth > 12 || ++nodes > 2000)
      throw new HttpError(
        400,
        "Payload nesting or complexity exceeds the supported limit",
      );
    if (value && typeof value === "object")
      for (const child of Object.values(value))
        pending.push({ value: child, depth: depth + 1 });
  }
  if (Buffer.byteLength(JSON.stringify(input) ?? "") > 32768)
    throw new HttpError(413, "Request is too large");
}
export class RequestLimits {
  private active = 0;
  private rates = new Map<string, { since: number; count: number }>();
  constructor(
    private maxConcurrent = 8,
    private perMinute = 120,
  ) {}
  enter(identity: string) {
    const now = Date.now();
    const entry = this.rates.get(identity);
    const rate =
      entry && now - entry.since < 60000 ? entry : { since: now, count: 0 };
    this.rates.set(identity, rate);
    if (++rate.count > this.perMinute)
      throw new HttpError(429, "Agent request rate exceeded");
    if (this.active >= this.maxConcurrent)
      throw new HttpError(429, "Gateway concurrency limit reached");
    this.active++;
    let released = false;
    return () => {
      if (!released) {
        released = true;
        this.active--;
      }
    };
  }
}

import { randomBytes, timingSafeEqual } from "node:crypto";
import { z } from "zod";

export type OperatorRole = "administrator" | "operator" | "approver" | "viewer";
export type OperatorIdentity = {
  id: string;
  name: string;
  tenant: string;
  role: OperatorRole;
};
export type OperatorCredential = OperatorIdentity & { key: string };
const operatorSchema = z
  .object({
    id: z.string().regex(/^[a-zA-Z0-9_-]{1,64}$/),
    name: z.string().min(1).max(80),
    tenant: z.string().regex(/^[a-zA-Z0-9_-]{1,64}$/),
    role: z.enum(["administrator", "operator", "approver", "viewer"]),
    key: z.string().min(24).max(200),
  })
  .strict();
const equal = (a: string, b: string) =>
  timingSafeEqual(Buffer.from(digest(a)), Buffer.from(digest(b)));
type Session = {
  actor: OperatorIdentity;
  expiresAt: number;
  csrfToken: string;
};
export class LocalOperators {
  private identities: { actor: OperatorIdentity; keyHash: string }[];
  private sessions = new Map<string, Session>();
  readonly team: boolean;
  constructor(adminKey: string, configured?: OperatorCredential[]) {
    this.team = configured !== undefined;
    const operators = z
      .array(operatorSchema)
      .min(1)
      .max(32)
      .parse(
        configured ?? [
          {
            id: "local-administrator",
            name: "Local owner",
            tenant: "local",
            role: "administrator",
            key: adminKey,
          },
        ],
      );
    if (
      new Set(operators.map((o) => o.id)).size !== operators.length ||
      new Set(operators.map((o) => digest(o.key))).size !== operators.length
    )
      throw Error("Operator identities and credentials must be unique");
    if (
      !operators.some((o) => o.tenant === "local" && o.role === "administrator")
    )
      throw Error("A local administrator is required");
    this.identities = operators.map(({ key, ...actor }) => ({
      actor,
      keyHash: digest(key),
    }));
  }
  login(key: string) {
    const hashed = digest(key);
    const matched = this.identities.find((o) => equal(o.keyHash, hashed));
    if (!matched) throw new HttpError(401, "Invalid operator credential");
    if (matched.actor.tenant !== "local")
      throw new HttpError(403, "Tenant access denied");
    for (const [id, session] of this.sessions)
      if (session.expiresAt <= Date.now()) this.sessions.delete(id);
    if (this.sessions.size >= 256)
      throw new HttpError(429, "Session limit reached");
    const token = randomBytes(32).toString("hex");
    const session = {
      actor: matched.actor,
      expiresAt: Date.now() + 8 * 3600_000,
      csrfToken: randomBytes(32).toString("hex"),
    };
    this.sessions.set(digest(token), session);
    return { token, ...session };
  }
  authenticate(cookie?: string) {
    const token = cookie
      ?.split(";")
      .map((x) => x.trim())
      .find((x) => x.startsWith("ab_session="))
      ?.slice(11);
    const session = token ? this.sessions.get(digest(token)) : undefined;
    if (
      !session ||
      session.expiresAt <= Date.now() ||
      session.actor.tenant !== "local"
    )
      throw new HttpError(401, "Operator sign-in required");
    return session;
  }
  checkCsrf(session: Session, token: unknown) {
    if (
      this.team &&
      (typeof token !== "string" || !equal(session.csrfToken, token))
    )
      throw new HttpError(403, "Session verification required");
  }
  logout(cookie?: string) {
    const token = cookie
      ?.split(";")
      .map((x) => x.trim())
      .find((x) => x.startsWith("ab_session="))
      ?.slice(11);
    if (token) this.sessions.delete(digest(token));
  }
}

// Only fixed server-owned JSON fields become SQL; every caller value is a parameter.
export function activityWhere(
  query: ActivityQuery,
  dialect: "sqlite" | "postgres",
) {
  const values: string[] = [],
    clauses: string[] = [];
  const field = (name: string) =>
    dialect === "sqlite"
      ? `json_extract(data,'$.${name}')`
      : `data::jsonb->>'${name}'`;
  const bind = (value: string) => {
    values.push(value);
    return dialect === "sqlite" ? "?" : "$" + values.length;
  };
  for (const key of [
    "agentId",
    "grantId",
    "tool",
    "status",
    "resource",
  ] as const)
    if (query[key]) clauses.push(`${field(key)}=${bind(query[key]!)}`);
  if (query.from) clauses.push(`${field("createdAt")}>=${bind(query.from)}`);
  if (query.to) clauses.push(`${field("createdAt")}<=${bind(query.to)}`);
  return {
    where: clauses.length ? " WHERE " + clauses.join(" AND ") : "",
    values,
  };
}

    };
    this.db
      .prepare("INSERT INTO agents VALUES (?,?,?)")
      .run(agent.id, digest(token), JSON.stringify(agent));
    this.audit("agent.created", { agentId: agent.id, name: agent.name });
    return { agent, token };
  }
  saveAgent(agent: Agent) {
    this.db
      .prepare("UPDATE agents SET data=? WHERE id=?")
      .run(JSON.stringify(agent), agent.id);
  }
  changeAgent(
    id: string,
    patch: { status?: Agent["status"]; grant?: TaskGrant },
    event: string,
  ) {
    this.db.exec("BEGIN IMMEDIATE");
    try {
      const agent = this.agent(id);
      if (agent) {
        if (patch.grant?.id === agent.grant?.id && patch.grant && agent.grant)
          patch = {
            ...patch,
            grant: {
              ...patch.grant,
              used_calls: Math.max(
                agent.grant.used_calls,
                patch.grant.used_calls,
              ),
            },
          };
        Object.assign(agent, patch);
        this.saveAgent(agent);
        this.audit(event, { agentId: id, grantId: agent.grant?.id });
      }
      this.db.exec("COMMIT");
      return agent;
    } catch (error) {
      this.db.exec("ROLLBACK");
      throw error;
    }
  }
  actions(): ActionRecord[] {
    return this.db
      .prepare("SELECT data FROM actions ORDER BY rowid DESC")
      .all()
      .map((r) => JSON.parse(r.data as string));
  }
  action(id: string): ActionRecord | undefined {
    const row = this.db.prepare("SELECT data FROM actions WHERE id=?").get(id);
    return row ? JSON.parse(row.data as string) : undefined;
  }
  previous(agentId: string, requestId: string): ActionRecord | undefined {
    const row = this.db
      .prepare("SELECT data FROM actions WHERE agent_id=? AND request_id=?")
      .get(agentId, requestId);
    return row ? JSON.parse(row.data as string) : undefined;
  }
  insertAction(a: ActionRecord) {
    this.db
      .prepare("INSERT INTO actions VALUES (?,?,?,?,?)")
      .run(a.id, a.agentId, a.requestId, a.status, JSON.stringify(a));
  }
  saveAction(a: ActionRecord) {
    this.db
      .prepare("UPDATE actions SET status=?, data=? WHERE id=?")
      .run(a.status, JSON.stringify(a), a.id);
  }
  claim(a: ActionRecord) {
    const oldStatus = a.status;
    a.status = "executing";
    return (
      this.db
        .prepare("UPDATE actions SET status=?, data=? WHERE id=? AND status=?")
        .run(a.status, JSON.stringify(a), a.id, oldStatus).changes === 1
    );
  }
  blockPending(a: ActionRecord) {
    this.db.exec("BEGIN IMMEDIATE");
    try {
      const persisted = this.action(a.id);
      if (!persisted || persisted.status !== "pending")
        throw new HttpError(409, "This action is no longer awaiting approval");
      const next = {
        ...persisted,
        status: "blocked" as const,
        rule: a.rule,
        reason: a.reason,
        approvalReceipt: a.approvalReceipt,
      };
      this.saveAction(next);
      this.audit("approval.blocked", { actionId: a.id, rule: a.rule });
      this.db.exec("COMMIT");
      Object.assign(a, next);
    } catch (error) {
      this.db.exec("ROLLBACK");
      throw error;
    }
  }
  reserve(a: ActionRecord, check: ReservationCheck) {
    this.db.exec("BEGIN IMMEDIATE");
    try {
      const persisted = this.action(a.id),
        agent = this.agent(a.agentId);
      if (
        !persisted ||
        !agent ||
        !["queued", "pending"].includes(persisted.status) ||
        persisted.status !== a.status
      )
        throw new HttpError(409, "Execution has already been claimed");
      const denial = check(agent, this.settings(), persisted);
      if (denial) {
        Object.assign(a, denial, { status: "blocked" });
        this.saveAction(a);
        this.audit("execution.blocked", { actionId: a.id, rule: denial.rule });
      } else {
        a.status = "executing";
        agent.grant!.used_calls++;
        if (a.tool === "database_read") agent.tainted = true;
        this.saveAgent(agent);
        this.saveAction(a);
        this.audit("execution.started", {
          actionId: a.id,
          correlationId: a.correlationId,
          approverId: a.approvalReceipt?.actorId,
        });
      }
      this.db.exec("COMMIT");
      return !denial;
    } catch (error) {
      this.db.exec("ROLLBACK");
      throw error;
    }
  }
  audit(type: string, details: unknown) {
    const previous =
      (this.db.prepare("SELECT hash FROM audit ORDER BY seq DESC LIMIT 1").get()
        ?.hash as string) ?? "genesis";
    const data = JSON.stringify({
      type,
      details,
      at: new Date().toISOString(),
    });
    this.db
      .prepare("INSERT INTO audit(data,previous_hash,hash) VALUES (?,?,?)")
      .run(data, previous, digest(previous + data));
  }
  auditTrail(query?: AuditQuery) {
    const where = query?.actionId
      ? " WHERE json_extract(data,'$.details.actionId')=?"
      : "";
    const values: (string | number)[] = query?.actionId ? [query.actionId] : [];
    if (query) values.push(query.limit, query.offset ?? 0);
    return this.db
      .prepare(
        "SELECT * FROM audit" +
          where +
          " ORDER BY seq" +
          (query ? " LIMIT ? OFFSET ?" : ""),
      )
      .all(...values) as {
      seq: number;
      data: string;
      previous_hash: string;
      hash: string;
    }[];
  }
  verifyAudit() {
    let previous = "genesis";
    for (const row of this.auditTrail()) {
      if (
        row.previous_hash !== previous ||
        row.hash !== digest(previous + row.data)
      )
        return false;
      previous = row.hash as string;
    }
    return true;
  }
  close() {
    this.db.close();
  }
}

import type express from "express";
import { SpanKind, SpanStatusCode } from "@opentelemetry/api";
import {
  TracerProvider,
  SimpleSpanProcessor,
  type SpanExporter,
} from "@opentelemetry/sdk-trace";
// Explicit SDK/exporter configuration: no automatic instrumentation, remote endpoint,
// environment resource detection, URL/body/header/error collection or user attributes.
export class LocalTelemetry {
  private provider: TracerProvider;
  constructor(
    private component: "gateway" | "control",
    write: (line: string) => void,
  ) {
    const exporter: SpanExporter = {
      export: (spans, callback) => {
        try {
          for (const span of spans) {
            const context = span.spanContext();
            const ns = (time: [number, number]) =>
              (BigInt(time[0]) * 1000000000n + BigInt(time[1])).toString();
            const attributes = Object.entries(span.attributes)
              .filter(([key]) =>
                [
                  "agentbrake.correlation_id",
                  "http.request.method",
                  "http.route",
                  "http.response.status_code",
                ].includes(key),
              )
              .map(([key, value]) => ({
                key,
                value:
                  typeof value === "number"
                    ? { intValue: String(value) }
                    : { stringValue: String(value) },
              }));
            write(
              JSON.stringify({
                resourceSpans: [
                  {
                    resource: {
                      attributes: [
                        {
                          key: "service.name",
                          value: {
                            stringValue: "agentbrake-" + this.component,
                          },
                        },
                      ],
                    },
                    scopeSpans: [
                      {
                        scope: { name: "agentbrake.local", version: "1" },
                        spans: [
                          {
                            traceId: context.traceId,
                            spanId: context.spanId,
                            name: "agentbrake.http",
                            kind: 2,
                            startTimeUnixNano: ns(span.startTime),
                            endTimeUnixNano: ns(span.endTime),
                            attributes,
                            status: { code: span.status.code },
                          },
                        ],
                      },
                    ],
                  },
                ],
              }),
            );
          }
          callback({ code: 0 });
        } catch {
          callback({ code: 1 });
        }
      },
      shutdown: async () => {},
    };
    this.provider = new TracerProvider({
      spanLimits: {
        attributeCountLimit: 8,
        attributeValueLengthLimit: 150,
        eventCountLimit: 0,
        linkCountLimit: 0,
      },
      spanProcessors: [new SimpleSpanProcessor({ exporter })],
    });
  }
  middleware(): express.RequestHandler {
    return (req, res, next) => {
      const span = this.provider
        .getTracer("agentbrake.local", "1")
        .startSpan("agentbrake.http", {
          kind: SpanKind.SERVER,
          attributes: {
            "agentbrake.correlation_id": res.locals.correlationId,
            "http.request.method": [
              "GET",
              "POST",
              "PUT",
              "PATCH",
              "DELETE",
              "HEAD",
              "OPTIONS",
            ].includes(req.method)
              ? req.method
              : "OTHER",
          },
        });
      res.setHeader("X-Trace-ID", span.spanContext().traceId);
      res.once("finish", () => {
        span.setAttribute("http.response.status_code", res.statusCode);
        span.setAttribute(
          "http.route",
          typeof req.route?.path === "string" ? req.route.path : "unmatched",
        );
        if (res.statusCode >= 500)
          span.setStatus({ code: SpanStatusCode.ERROR });
        span.end();
      });
      next();
    };
  }
  shutdown() {
    return this.provider.shutdown();
  }
}
